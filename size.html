2  Size portfolios

This pilot forms monthly size quintiles and measures their next-month returns. It uses CRSP CIZ, a short historical subset (2000–2002 holding months), and breakpoints computed from all eligible observations rather than NYSE-only breakpoints. Its purpose is to compare the Polars sort with AnomalyLab on identical inputs, not to match the textbook’s tables.

2.1 Read the stock-month panel

Use ordinary U.S.-incorporated common shares of corporations on the NYSE, AMEX, and Nasdaq at formation. Use positive market capitalization for both the sort and the portfolio weights. Don’t require a stock to meet the same screen in the holding month: keep its reported CIZ return if present, including a return for a delisting month when available.

import polars as pl
from era_pl import load_parquet

first = pl.date(2000, 1, 1)
last = pl.date(2002, 12, 1)
monthly = (
    load_parquet("msf_v2", "crsp")
    .with_columns(month=pl.col("mthcaldt").dt.month_start())
    .filter(pl.col("month").is_between(first.dt.offset_by("-1mo"), last))
)
stock_months = monthly.select("permno", "month")
assert stock_months.collect().is_duplicated().sum() == 0, "Duplicate stock-months"

formation = (
    monthly.filter(
        pl.col("securitytype") == "EQTY",
        pl.col("securitysubtype") == "COM",
        pl.col("sharetype") == "NS",
        pl.col("usincflg") == "Y",
        pl.col("issuertype").is_in(["ACOR", "CORP"]),
        pl.col("primaryexch").is_in(["A", "N", "Q"]),
        pl.col("tradingstatusflg") == "A",
        pl.col("conditionaltype") == "RW",
        pl.col("mthcap") > 0,
    )
    .select(
        "permno",
        pl.col("month").dt.offset_by("1mo").alias("month"),
        pl.col("mthcap").alias("size"),
    )
    .filter(pl.col("month").is_between(first, last))
)
returns = monthly.select("permno", "month", pl.col("mthret").alias("ret"))
panel = formation.join(returns, on=["permno", "month"], how="left").collect()
complete = panel.filter(pl.col("ret").is_finite())
print(f"{panel.height:,} formation observations; {complete.height:,} with a usable return.")
218,291 formation observations; 215,991 with a usable return.

The join advances the calendar by exactly one month. It does not jump a gap to the next available observed return. For this parity check, we follow AnomalyLab by dropping missing holding-period returns before computing breakpoints. That can change the portfolios using information that wasn’t known at formation. A backtest should instead freeze the breakpoints and assignments at formation, then handle missing returns separately.

2.2 Sort within each month

Use linear quantiles and right-closed bins, like the reference’s pandas.qcut quintiles. Values equal to a breakpoint stay in the lower bin. Stop rather than silently breaking ties if quantile edges coincide.

probs = [0, .2, .4, .6, .8, 1]
edges = [f"q{i}" for i in range(len(probs))]
cutoffs = complete.group_by("month").agg(
    pl.col("size").quantile(q, interpolation="linear").alias(edge)
    for q, edge in zip(probs, edges)
)
assert cutoffs.height > 0, "No complete stock-months"
assert not cutoffs.select(
    pl.any_horizontal(pl.col(a) >= pl.col(b) for a, b in zip(edges, edges[1:])).any()
).item(), "Non-unique quantile edges"

members = (
    complete.join(cutoffs, on="month")
    .with_columns(
        quintile=1 + pl.sum_horizontal(
            (pl.col("size") > pl.col(edge)).cast(pl.Int32) for edge in edges[1:-1]
        )
    )
    .select("permno", "month", "size", "ret", "quintile")
)

2.3 Compare returns

Compute one equal-weighted and one value-weighted return per month and quintile. Then take the arithmetic mean over months. The spread is large minus small. Returns in the table are percentages per month; the calculations use decimals.

monthly_returns = members.group_by("month", "quintile").agg(
    equal_weighted=pl.col("ret").mean(),
    value_weighted=(pl.col("ret") * pl.col("size")).sum() / pl.col("size").sum(),
).sort("month", "quintile")
assert monthly_returns.group_by("month").len()["len"].eq(5).all()

small = monthly_returns.filter(pl.col("quintile") == 1)
large = monthly_returns.filter(pl.col("quintile") == 5)
spread = large.join(small, on="month", suffix="_small").select(
    "month",
    pl.lit(6, dtype=pl.Int32).alias("quintile"),
    (pl.col("equal_weighted") - pl.col("equal_weighted_small")).alias("equal_weighted"),
    (pl.col("value_weighted") - pl.col("value_weighted_small")).alias("value_weighted"),
)
all_returns = pl.concat([monthly_returns, spread])
summary = all_returns.group_by("quintile").agg(
    pl.len().alias("months"),
    (100 * pl.col("equal_weighted").mean()).round(3).alias("equal_weighted_pct"),
    (100 * pl.col("value_weighted").mean()).round(3).alias("value_weighted_pct"),
).sort("quintile").with_columns(
    portfolio=pl.when(pl.col("quintile") == 6).then(pl.lit("Large − small"))
    .otherwise(pl.col("quintile").cast(pl.String))
).drop("quintile").select("portfolio", "months", "equal_weighted_pct", "value_weighted_pct")

2.4 Risk-adjusted returns

A portfolio’s average return includes compensation for exposures to common risk factors. We estimate Fama–French three-factor alphas by regressing each portfolio’s excess return on the market excess return, SMB, and HML. Subtract the risk-free rate from each quintile’s return. For the long–short spread the two risk-free rates cancel.

factors = (
    load_parquet("factors_monthly", "ff")
    .with_columns(month=pl.col("date").dt.month_start())
    .select("month", "mktrf", "smb", "hml", "rf")
    .filter(pl.col("month").is_between(first, last))
    .collect()
)
factor_data = (
    all_returns
    .join(factors, on="month", validate="m:1")
    .unpivot(
        on=["equal_weighted", "value_weighted"],
        index=["month", "quintile", "mktrf", "smb", "hml", "rf"],
        variable_name="weighting",
        value_name="ret",
    )
    .with_columns(
        excess=pl.when(pl.col("quintile") == 6)
        .then(pl.col("ret"))
        .otherwise(pl.col("ret") - pl.col("rf"))
    )
    .sort("month")
)

The intercept is monthly alpha. We use Bartlett Newey–West standard errors with three lags and no small-sample correction, matching the numeric AnomalyLab calculation. The regressions use only 36 months, so these are illustrations of the method rather than a test of the size premium.

import statsmodels.formula.api as smf

assert factor_data.height == 2 * all_returns.height
assert not any(factor_data.select("mktrf", "smb", "hml", "rf").null_count().row(0))

fits = {
    key: smf.ols("excess ~ mktrf + smb + hml", data=series.to_pandas()).fit(
        cov_type="HAC", cov_kwds={"maxlags": 3, "use_correction": False}
    )
    for key, series in factor_data.group_by("quintile", "weighting")
}
alphas = pl.DataFrame([
    {"quintile": quintile, "weighting": weighting, "months": int(fm.nobs),
     "alpha_pct": 100 * fm.params["Intercept"], "t": fm.tvalues["Intercept"]}
    for (quintile, weighting), fm in fits.items()
]).sort("quintile", "weighting")
alphas.with_columns(
    portfolio=pl.when(pl.col("quintile") == 6).then(pl.lit("Large − small"))
    .otherwise(pl.col("quintile").cast(pl.String)),
    alpha_pct=pl.col("alpha_pct").round(3), t=pl.col("t").round(3),
).select("portfolio", "weighting", "months", "alpha_pct", "t")
